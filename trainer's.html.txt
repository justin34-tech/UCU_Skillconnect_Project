  <script src="https://cdn.jsdelivr.net/npm/bootstrap@5.3.3/dist/js/bootstrap.bundle.min.js"></script>
  <script>
    // Sample skills displayed when the page first opens.
    let skills = [
      {name:"Python Programming", category:"Programming", level:"Beginner", trainer:"Juliet Laura"},
      {name:"Data Analysis", category:"Data", level:"Intermediate", trainer:"Abalo Lucy"},
      {name:"Web Development", category:"Programming", level:"Beginner", trainer:"Grace Mbabazi"},
      {name:"Cybersecurity", category:"Cybersecurity", level:"Intermediate", trainer:"Mr. Timoty"},
      {name:"Graphic Design", category:"Design", level:"Beginner", trainer:"Mr. Justin"},
      {name:"HTML & CSS", category:"Programming", level:"Beginner", trainer:"Mr. Nsaawa"},
      {name:"JavaScript", category:"Programming", level:"Intermediate", trainer:"Mr. Jonathan"},
      {name:"Machine Learning", category:"Data", level:"Advanced", trainer:"Mr. Kigozi"}
    ];

    // Add trainer skills saved in this browser to the displayed skill list.
    let registeredTrainers = [];
    try {
      registeredTrainers = JSON.parse(localStorage.getItem("skillConnectTrainers") || "[]");
      if (!Array.isArray(registeredTrainers)) registeredTrainers = [];
    } catch (error) {
      registeredTrainers = [];
    }
    registeredTrainers.forEach(trainer => {
      (trainer.skills || []).forEach(skillName => {
        skills.push({name:skillName, category:guessCategory(skillName), level:"Beginner", trainer:trainer.name});
      });
    });

    function guessCategory(skillName) {
      const value = skillName.toLowerCase();
      if (value.includes("data") || value.includes("machine learning")) return "Data";
      if (value.includes("design")) return "Design";
      if (value.includes("cyber")) return "Cybersecurity";
      if (value.includes("network")) return "Networking";
      if (["python","javascript","java","html","css","web","programming"].some(word => value.includes(word))) return "Programming";
      return "Other";
    }

    function showSkills() {
      const word = document.getElementById("search").value.trim().toLowerCase();
      const cat = document.getElementById("category").value;
      const lev = document.getElementById("level").value;
      const found = skills.filter(skill =>
        skill.name.toLowerCase().includes(word) &&
        (!cat || skill.category === cat) &&
        (!lev || skill.level === lev)
      );

      document.getElementById("results").innerHTML = found.map(skill => `
        <div class="col-md-6 col-lg-4">
          <div class="card skill-card"><div class="card-body p-4">
            <div class="icon-box mb-3">✦</div>
            <h5 class="fw-bold">${escapeHTML(skill.name)}</h5>
            <p class="text-muted small mb-2">Trainer: ${escapeHTML(skill.trainer)}</p>
            <span class="tag">${escapeHTML(skill.category)}</span>
          </div></div>
        </div>`).join("");

      document.getElementById("empty").classList.toggle("d-none", found.length > 0);
      renderOffers();
    }

    function renderOffers() {
      document.getElementById("offers").innerHTML = skills.map(skill => `
        <div class="col-md-6 col-lg-4">
          <div class="card trainer-card"><div class="card-body p-4">
            <div class="d-flex align-items-center gap-3 mb-3">
              <div class="icon-box">✓</div>
              <div><h6 class="fw-bold mb-0">${escapeHTML(skill.name)}</h6><small class="text-muted">Trainer: ${escapeHTML(skill.trainer)}</small></div>
            </div>
            <span class="tag">${escapeHTML(skill.category)}</span>
          </div></div>
        </div>`).join("");
      document.getElementById("noOffers").classList.toggle("d-none", skills.length > 0);
    }

    function escapeHTML(value) {
      return String(value).replace(/[&<>"']/g, character => ({
        "&":"&amp;", "<":"&lt;", ">":"&gt;", '"':"&quot;", "'":"&#39;"
      })[character]);
    }

    function goSearch() {
      document.getElementById("search").value = document.getElementById("homeSearch").value;
      showSkills();
      document.getElementById("find").scrollIntoView({behavior:"smooth"});
    }

    // Register the trainer and display their skills. The password is never stored.
    document.getElementById("registerForm").addEventListener("submit", function(event) {
      event.preventDefault();
      const name = document.getElementById("trainerName").value.trim();
      const email = document.getElementById("trainerEmail").value.trim().toLowerCase();
      const password = document.getElementById("trainerPassword").value;
      const skillNames = document.getElementById("trainerSkills").value
        .split(",").map(item => item.trim()).filter(Boolean);
      const message = document.getElementById("registerMessage");

      if (password.length < 6) {
        message.className = "mt-3 text-danger";
        message.textContent = "Please use a password with at least 6 characters.";
        return;
      }
      if (skillNames.length === 0) {
        message.className = "mt-3 text-danger";
        message.textContent = "Please enter at least one skill you can teach.";
        return;
      }
      if (registeredTrainers.some(trainer => trainer.email === email)) {
        message.className = "mt-3 text-danger";
        message.textContent = "This email has already been registered on this browser.";
        return;
      }

      const trainer = {name:name, email:email, skills:skillNames};
      registeredTrainers.push(trainer);
      try {
        localStorage.setItem("skillConnectTrainers", JSON.stringify(registeredTrainers));
      } catch (error) {
        registeredTrainers.pop();
        message.className = "mt-3 text-danger";
        message.textContent = "Your browser could not save this registration. Please check your browser storage settings.";
        return;
      }

      skillNames.forEach(skillName => {
        skills.push({name:skillName, category:guessCategory(skillName), level:"Beginner", trainer:name});
      });
      showSkills();
      this.reset();
      message.className = "alert alert-success mt-3 mb-0";
      message.textContent = "Registration successful! Your trainer skills have been added to Skills Offered.";
      const registrationModal = bootstrap.Modal.getOrCreateInstance(document.getElementById("registerModal"));
      registrationModal.hide();
      document.getElementById("offered").scrollIntoView({behavior:"smooth"});
    });

    showSkills();
  </script>
