<script src="https://cdn.jsdelivr.net/npm/bootstrap@5.3.3/dist/js/bootstrap.bundle.min.js"></script>
<script>
  // Skill data used by both skill lists.
  let skills = [
    ["Python Programming","Programming","Beginner","Juliet Laura"],
    ["Data Analysis","Data","Intermediate","Abalo Lucy"],
    ["Web Development","Programming","Beginner","Grace Mbabazi"],
    ["Cybersecurity","Cybersecurity","Intermediate","Mr. Timoty"],
    ["Graphic Design","Design","Beginner","Mr. Justin"],
    ["HTML & CSS","Programming","Beginner","Mr. Nsaawa"],
    ["JavaScript","Programming","Intermediate","Mr. Jonathan"],
    ["Machine Learning","Data","Advanced","Mr. Kigozi"]
  ];

  // Show skills that match the search, category and level.
  function showSkills(){
    let word = search.value.toLowerCase(), cat = category.value, lev = level.value;
    let found = skills.filter(s => s[0].toLowerCase().includes(word) && (!cat || s[1]===cat) && (!lev || s[2]===lev));
    let cards = found.map(s => `<div class="col-md-4"><div class="card skill-card"><div class="card-body"><h5>${s[0]}</h5><p>Trainer: ${s[3]}</p></div></div></div>`).join("");
    results.innerHTML = cards; offers.innerHTML = skills.map(s => `<div class="col-md-4"><div class="card"><div class="card-body"><b>${s[0]}</b><p>Trainer: ${s[3]}</p></div></div></div>`).join("");
    empty.classList.toggle("d-none", found.length > 0);
  }

  // Copy the home search into Find Skills and display the results.
  function goSearch(){ search.value = homeSearch.value; showSkills(); document.getElementById("find").scrollIntoView({behavior:"smooth"}); }

  // Add a new skill to the array and refresh the lists.
  skillForm.onsubmit = function(e){
    e.preventDefault();
    skills.push([skillName.value,skillCategory.value,skillLevel.value,trainer.value]);
    showSkills(); this.reset(); notice.textContent = "Skill added!";
  };

  // Save a simple demo account in this browser.
  registerForm.onsubmit = function(e){
    e.preventDefault();
    localStorage.setItem("skillAccount",JSON.stringify({name:name.value,email:email.value.toLowerCase(),role:role.value}));
    registerMessage.textContent = "Account saved. You can now try logging in.";
    this.reset();
  };

  // Check the email and role saved during registration.
  loginForm.onsubmit = function(e){
    e.preventDefault();
    let account = JSON.parse(localStorage.getItem("skillAccount") || "null");
    if(account && account.email===loginEmail.value.toLowerCase() && account.role===loginRole.value){
      loginMessage.textContent = "Welcome, "+account.name+"! You are logged in as a "+account.role+".";
      loginMessage.className = "text-success mt-2";
      welcome.textContent = "Welcome, "+account.name+" ("+account.role+")";
    }else{
      loginMessage.textContent = account ? "Email or role does not match." : "Please register first.";
      loginMessage.className = "text-danger mt-2";
    }
  };

  showSkills(); // Display the starting skills when the page loads.
</script>
